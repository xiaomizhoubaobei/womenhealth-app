## 一、功能概述

### 1.1 功能定义
周期变化趋势分析功能通过对用户历史周期数据的深度分析，提供周期规律性评估、趋势可视化、异常预警和健康风险评估，帮助用户全面了解自身健康状况。

### 1.2 核心特性
- **规律性评估**：基于统计学方法评估周期规律性，给出评分和建议
- **趋势可视化**：使用图表展示周期长度、症状强度等变化趋势
- **异常检测**：自动识别异常周期，提供预警和就医建议
- **健康评估**：基于周期数据评估潜在健康风险（如PCOS、贫血等）
- **个性化建议**：根据分析结果提供针对性的健康建议

### 1.3 技术亮点
- 本地计算，保护用户隐私
- 多维度分析（周期、症状、健康风险）
- 智能趋势识别（线性回归算法）
- 可视化数据展示（图表、统计卡片）

### 步骤2: 创建趋势分析算法引擎

**文件路径**: `domain/TrendAnalysisEngine.kt`
```kotlin
package com.example.womenhealthapp.domain

import com.example.womenhealthapp.data.entity.MenstrualCycle
import com.example.womenhealthapp.data.model.*
import kotlin.math.abs
import kotlin.math.sqrt

/**
 * 周期变化趋势分析引擎
 * 负责计算周期规律性、趋势、异常检测和健康风险评估
 */
class TrendAnalysisEngine(
    private val minCyclesForAnalysis: Int = 4,
    private val maxCyclesForAnalysis: Int = 12
) {

    /**
     * 执行完整的趋势分析
     */
    fun analyze(cycles: List<MenstrualCycle>): TrendAnalysisResult {
        // 筛选有效周期
        val validCycles = cycles.filter { it.cycleLength in 18..45 }
            .sortedBy { it.startDate }

        val regularityScore = if (validCycles.size >= 3) {
            calculateRegularityScore(validCycles)
        } else null

        val trendMetrics = if (validCycles.size >= 4) {
            calculateTrendMetrics(validCycles)
        } else null

        val anomalies = if (validCycles.size >= 4) {
            detectAnomalies(validCycles)
        } else emptyList()

        val healthRisks = if (validCycles.size >= 4) {
            assessHealthRisks(validCycles)
        } else emptyList()

        return TrendAnalysisResult(
            regularityScore = regularityScore,
            trendMetrics = trendMetrics,
            anomalies = anomalies,
            healthRisks = healthRisks
        )
    }

    /**
     * 计算周期规律性评分
     */
    private fun calculateRegularityScore(cycles: List<MenstrualCycle>): CycleRegularityScore {
        val cycleLengths = cycles.map { it.cycleLength.toFloat() }
        val n = cycleLengths.size

        // 计算平均值
        val mean = cycleLengths.average()
        val avgCycleLength = mean.toFloat()

        // 计算标准差
        val variance = cycleLengths.map { (it - mean).pow(2) }.average()
        val stdDeviation = sqrt(variance).toFloat()

        // 计算变异系数
        val coefficientOfVariation = if (avgCycleLength > 0) {
            stdDeviation / avgCycleLength
        } else 0f

        // 规律性评级
        val level = RegularityLevel.fromCV(coefficientOfVariation)

        // 计算评分
        val score = calculateScoreFromCV(coefficientOfVariation, level)

        // 检测异常周期（超出2个标准差）
        val lowerBound = mean - 2 * stdDeviation
        val upperBound = mean + 2 * stdDeviation
        val outlierCycleIds = cycles.filter {
            it.cycleLength < lowerBound || it.cycleLength > upperBound
        }.map { it.id }

        // 生成洞察
        val insights = generateRegularityInsights(level, score, outlierCycleIds)

        return CycleRegularityScore(
            score = score,
            level = level,
            avgCycleLength = avgCycleLength,
            stdDeviation = stdDeviation,
            coefficientOfVariation = coefficientOfVariation,
            range = cycleLengths.minOrNull()?.toInt()!!..cycleLengths.maxOrNull()?.toInt()!!,
            outlierCycleIds = outlierCycleIds,
            insights = insights
        )
    }

    /**
     * 根据变异系数计算评分
     */
    private fun calculateScoreFromCV(cv: Float, level: RegularityLevel): Float {
        return when (level) {
            RegularityLevel.VERY_REGULAR -> 90 + (1 - cv / 0.05f) * 10
            RegularityLevel.REGULAR -> 70 + (1 - cv / 0.10f) * 20
            RegularityLevel.SOMEWHAT_IRREGULAR -> 50 + (1 - cv / 0.15f) * 20
            RegularityLevel.IRREGULAR -> (50 - (cv - 0.15f) * 100).coerceAtLeast(0f)
        }.coerceIn(0f, 100f)
    }

    /**
     * 生成规律性洞察
     */
    private fun generateRegularityInsights(
        level: RegularityLevel,
        score: Float,
        outlierCycleIds: List<Long>
    ): List<String> {
        val insights = mutableListOf<String>()

        when (level) {
            RegularityLevel.VERY_REGULAR -> {
                insights.add("您的月经周期非常规律，健康状况良好")
                insights.add("规律周期是身体健康的重要指标")
            }
            RegularityLevel.REGULAR -> {
                insights.add("您的月经周期比较规律，基本正常")
                insights.add("继续保持良好的生活习惯")
            }
            RegularityLevel.SOMEWHAT_IRREGULAR -> {
                insights.add("您的月经周期有些不规律，建议关注")
                insights.add("可能是压力、饮食或睡眠变化导致的")
                if (outlierCycleIds.isNotEmpty()) {
                    insights.add("检测到${outlierCycleIds.size}个异常周期，建议记录更多细节")
                }
            }
            RegularityLevel.IRREGULAR -> {
                insights.add("您的月经周期明显不规律，建议就医咨询")
                insights.add("周期不规律可能与多种因素有关，需要专业评估")
                insights.add("建议记录症状、饮食、运动等详细信息")
            }
        }

        return insights
    }

    /**
     * 计算周期趋势指标（线性回归）
     */
    private fun calculateTrendMetrics(cycles: List<MenstrualCycle>): CycleTrendMetrics {
        // 生成时间序列（使用索引作为x值）
        val data = cycles.mapIndexed { index, cycle ->
            Pair(index.toFloat(), cycle.cycleLength.toFloat())
        }

        // 线性回归
        val regression = linearRegression(data)

        // 判断趋势方向
        val trendDirection = determineTrendDirection(regression.slope, regression.rSquared)

        // 计算变化率
        val changeRate = regression.slope // 每个周期的变化天数

        // 评估显著性
        val significance = assessSignificance(regression.rSquared, data.size)

        // 生成趋势洞察
        val insights = generateTrendInsights(regression, trendDirection, changeRate, significance)

        return CycleTrendMetrics(
            slope = regression.slope,
            rSquared = regression.rSquared,
            trendDirection = trendDirection,
            changeRate = changeRate,
            significance = significance,
            insights = insights
        )
    }

    /**
     * 线性回归算法
     * 返回斜率和拟合度（R²）
     */
    private fun linearRegression(data: List<Pair<Float, Float>>): RegressionResult {
        val n = data.size
        val sumX = data.sumOf { it.first.toDouble() }
        val sumY = data.sumOf { it.second.toDouble() }
        val sumXY = data.sumOf { it.first.toDouble() * it.second.toDouble() }
        val sumX2 = data.sumOf { it.first.toDouble() * it.first.toDouble() }
        val sumY2 = data.sumOf { it.second.toDouble() * it.second.toDouble() }

        val denominator = n * sumX2 - sumX * sumX
        val slope = if (denominator != 0.0) {
            (n * sumXY - sumX * sumY) / denominator
        } else 0.0

        val intercept = if (denominator != 0.0) {
            (sumY - slope * sumX) / n
        } else sumY / n

        // 计算R²（拟合度）
        val meanY = sumY / n
        val ssTotal = data.sumOf { (it.second - meanY) * (it.second - meanY) }
        val ssResidual = data.sumOf {
            val predicted = slope * it.first + intercept
            (it.second - predicted) * (it.second - predicted)
        }
        val rSquared = if (ssTotal != 0.0) 1 - ssResidual / ssTotal else 0.0

        return RegressionResult(
            slope = slope.toFloat(),
            intercept = intercept.toFloat(),
            rSquared = rSquared.toFloat()
        )
    }

    /**
     * 判断趋势方向
     */
    private fun determineTrendDirection(slope: Float, rSquared: Float): TrendDirection {
        val significance = rSquared > 0.5f

        return when {
            !significance || abs(slope) < 0.3f -> TrendDirection.STABLE
            slope > 0.5f -> TrendDirection.INCREASING
            slope < -0.5f -> TrendDirection.DECREASING
            slope > 0 -> TrendDirection.INCREASING
            else -> TrendDirection.DECREASING
        }
    }

    /**
     * 评估趋势显著性
     */
    private fun assessSignificance(rSquared: Float, n: Int): TrendSignificance {
        // 考虑样本数量的调整
        val adjustedRSquared = if (n > 2) {
            rSquared * (n - 1) / (n - 2)
        } else rSquared

        return when {
            adjustedRSquared > 0.8f -> TrendSignificance.HIGH
            adjustedRSquared > 0.5f -> TrendSignificance.MODERATE
            adjustedRSquared > 0.3f -> TrendSignificance.LOW
            else -> TrendSignificance.NOT_SIGNIFICANT
        }
    }

    /**
     * 生成趋势洞察
     */
    private fun generateTrendInsights(
        regression: RegressionResult,
        trendDirection: TrendDirection,
        changeRate: Float,
        significance: TrendSignificance
    ): List<String> {
        val insights = mutableListOf<String>()

        when (trendDirection) {
            TrendDirection.STABLE -> {
                insights.add("您的周期长度保持稳定")
                if (significance == TrendSignificance.HIGH) {
                    insights.add("数据显示您的周期非常稳定")
                }
            }
            TrendDirection.INCREASING -> {
                val changeDesc = if (abs(changeRate) >= 1.0f) {
                    "${changeRate.toInt()}天/周期"
                } else {
                    "${(changeRate * 30).toInt()}天/年"
                }
                insights.add("您的周期长度在增加，平均每个周期增加${changeDesc}")
                if (significance == TrendSignificance.HIGH || significance == TrendSignificance.MODERATE) {
                    insights.add("这一趋势较为明显，建议关注")
                    insights.add("可能与年龄、激素变化或生活方式有关")
                }
            }
            TrendDirection.DECREASING -> {
                val changeDesc = if (abs(changeRate) >= 1.0f) {
                    "${abs(changeRate).toInt()}天/周期"
                } else {
                    "${(abs(changeRate) * 30).toInt()}天/年"
                }
                insights.add("您的周期长度在减少，平均每个周期减少${changeDesc}")
                if (significance == TrendSignificance.HIGH || significance == TrendSignificance.MODERATE) {
                    insights.add("这一趋势较为明显，建议关注")
                }
            }
            TrendDirection.FLUCTUATING -> {
                insights.add("您的周期长度波动较大")
                insights.add("可能与压力、饮食或睡眠变化有关")
            }
        }

        return insights
    }

    /**
     * 检测异常周期
     */
    private fun detectAnomalies(cycles: List<MenstrualCycle>): List<CycleAnomaly> {
        val anomalies = mutableListOf<CycleAnomaly>()

        if (cycles.size < 4) return anomalies

        // 计算基线统计（排除最后一个周期）
        val baselineCycles = cycles.dropLast(1)
        val cycleLengths = baselineCycles.map { it.cycleLength.toFloat() }
        val mean = cycleLengths.average()
        val variance = cycleLengths.map { (it - mean).pow(2) }.average()
        val stdDev = sqrt(variance)

        val lowerBound = mean - 2 * stdDev
        val upperBound = mean + 2 * stdDev

        // 检查每个周期
        cycles.forEach { cycle ->
            val length = cycle.cycleLength
            val deviation = abs(length - mean)
            val deviationPercentage = (deviation / mean * 100).toFloat()

            val anomalyType = when {
                length < lowerBound && deviationPercentage > 30f -> AnomalyType.TOO_SHORT
                length > upperBound && deviationPercentage > 30f -> AnomalyType.TOO_LONG
                length < mean - stdDev && deviationPercentage > 15f -> AnomalyType.SUDDEN_SHORTENING
                length > mean + stdDev && deviationPercentage > 15f -> AnomalyType.SUDDEN_LENGTHENING
                deviationPercentage > 25f -> AnomalyType.EXTREME_FLUCTUATION
                else -> null
            }

            anomalyType?.let { type ->
                val severity = when {
                    deviationPercentage > 30f -> AnomalySeverity.HIGH
                    deviationPercentage > 15f -> AnomalySeverity.MEDIUM
                    else -> AnomalySeverity.LOW
                }

                anomalies.add(
                    CycleAnomaly(
                        cycleId = cycle.id,
                        cycleDate = cycle.startDate,
                        anomalyType = type,
                        severity = severity,
                        expectedRange = lowerBound.toInt()..upperBound.toInt(),
                        actualValue = length,
                        deviation = deviation.toFloat(),
                        deviationPercentage = deviationPercentage,
                        possibleCauses = getPossibleCauses(type),
                        recommendations = getRecommendations(type, severity),
                        shouldConsultDoctor = severity == AnomalySeverity.HIGH
                    )
                )
            }
        }

        return anomalies
    }

    /**
     * 获取建议
     */
    private fun getRecommendations(
        anomalyType: AnomalyType,
        severity: AnomalySeverity
    ): List<String> {
        val recommendations = mutableListOf<String>()

        recommendations.add("记录更多周期数据以确认趋势")
        recommendations.add("保持规律的作息和饮食")
        recommendations.add("适当运动，但避免过度")

        if (severity == AnomalySeverity.HIGH) {
            recommendations.add("建议咨询妇科医生")
            recommendations.add("可能需要激素检查")
        }

        return recommendations
    }

    /**
     * 评估健康风险
     */
    private fun assessHealthRisks(cycles: List<MenstrualCycle>): List<HealthRiskAssessment> {
        val risks = mutableListOf<HealthRiskAssessment>()

        // 评估贫血风险
        val anemiaRisk = assessAnemiaRisk(cycles)
        if (anemiaRisk != null) {
            risks.add(anemiaRisk)
        }

        // 评估压力相关风险
        val stressRisk = assessStressRisk(cycles)
        if (stressRisk != null) {
            risks.add(stressRisk)
        }

        return risks
    }

    /**
     * 评估贫血风险
     */
    private fun assessAnemiaRisk(cycles: List<MenstrualCycle>): HealthRiskAssessment? {
        val heavyPeriods = cycles.count { it.flowIntensity == "heavy" }
        val totalPeriods = cycles.size
        val heavyPeriodRatio = heavyPeriods.toFloat() / totalPeriods

        var riskScore = 0
        if (heavyPeriodRatio > 0.5f) riskScore += 2
        if (heavyPeriodRatio > 0.3f) riskScore += 1

        val probability = minOf(1.0f, riskScore * 0.3f)

        if (probability < 0.3f) return null

        val riskLevel = when {
            probability > 0.6f -> RiskLevel.HIGH
            probability > 0.4f -> RiskLevel.MODERATE
            else -> RiskLevel.LOW
        }

        return HealthRiskAssessment(
            riskType = HealthRiskType.ANEMIA,
            riskLevel = riskLevel,
            probability = probability,
            evidence = listOf(
                if (heavyPeriodRatio > 0.3f) "经血量大（${(heavyPeriodRatio * 100).toInt()}%）" else ""
            ).filter { it.isNotEmpty() },
            recommendations = listOf(
                "建议进行血常规检查",
                "考虑补充铁剂",
                "增加富含铁的食物摄入"
            ),
            referenceLinks = listOf(
                "https://www.mayoclinic.org/diseases-conditions/iron-deficiency-anemia"
            )
        )
    }

    /**
     * 评估压力相关风险
     */
    private fun assessStressRisk(cycles: List<MenstrualCycle>): HealthRiskAssessment? {
        val cycleLengths = cycles.map { it.cycleLength.toFloat() }
        val mean = cycleLengths.average()
        val variance = cycleLengths.map { (it - mean).pow(2) }.average()
        val stdDev = sqrt(variance)
        val cv = if (mean > 0) stdDev / mean else 0.0

        if (cv < 0.10) return null // 规律周期，无压力风险

        val probability = when {
            cv > 0.20 -> 0.8f
            cv > 0.15 -> 0.6f
            cv > 0.10 -> 0.4f
            else -> 0.2f
        }

        val riskLevel = when {
            probability > 0.6f -> RiskLevel.HIGH
            probability > 0.4f -> RiskLevel.MODERATE
            else -> RiskLevel.LOW
        }

        return HealthRiskAssessment(
            riskType = HealthRiskType.STRESS_RELATED,
            riskLevel = riskLevel,
            probability = probability,
            evidence = listOf(
                "周期不规律（变异系数${String.format("%.1f%%", cv * 100)}）"
            ),
            recommendations = listOf(
                "建议适当减轻压力",
                "保证充足睡眠",
                "考虑进行放松训练或冥想",
                "如持续不规律，建议就医咨询"
            ),
            referenceLinks = emptyList()
        )
    }

    /**
     * 线性回归结果
     */
    private data class RegressionResult(
        val slope: Float,
        val intercept: Float,
        val rSquared: Float
    )
}
```

```kotlin
package com.example.womenhealthapp.viewmodel

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.example.womenhealthapp.data.model.TrendAnalysisResult
import com.example.womenhealthapp.repository.WomenHealthRepository
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class TrendAnalysisViewModel(
    private val repository: WomenHealthRepository
) : ViewModel() {

    private val _analysisResult = MutableStateFlow<TrendAnalysisResult?>(null)
    val analysisResult: StateFlow<TrendAnalysisResult?> = _analysisResult.asStateFlow()

    private val _isLoading = MutableStateFlow(false)
    val isLoading: StateFlow<Boolean> = _isLoading.asStateFlow()

    private val _error = MutableStateFlow<String?>(null)
    val error: StateFlow<String?> = _error.asStateFlow()

    /**
     * 加载趋势分析结果
     */
    fun loadTrendAnalysis() {
        viewModelScope.launch {
            try {
                _isLoading.value = true
                _error.value = null

                val result = repository.analyzeTrends()
                _analysisResult.value = result
            } catch (e: Exception) {
                _error.value = "趋势分析失败: ${e.message}"
            } finally {
                _isLoading.value = false
            }
        }
    }

    /**
     * 刷新分析
     */
    fun refreshAnalysis() {
        loadTrendAnalysis()
    }

    /**
     * 清除错误信息
     */
    fun clearError() {
        _error.value = null
    }
}
```

## 五、测试方案

### 5.1 单元测试

**文件路径**: `domain/TrendAnalysisEngineTest.kt`
```kotlin
package com.example.womenhealthapp.domain

import com.example.womenhealthapp.data.entity.MenstrualCycle
import com.example.womenhealthapp.data.model.*
import org.junit.Assert.*
import org.junit.Before
import org.junit.Test
import java.util.Calendar

class TrendAnalysisEngineTest {

    private lateinit var engine: TrendAnalysisEngine

    @Before
    fun setup() {
        engine = TrendAnalysisEngine()
    }

    @Test
    fun `analyze should return insufficient data when cycles less than 3`() {
        val cycles = listOf(
            createCycle(daysAgo = 28, cycleLength = 28),
            createCycle(daysAgo = 56, cycleLength = 29)
        )

        val result = engine.analyze(cycles)

        assertNull(result.regularityScore)
        assertNull(result.trendMetrics)
        assertTrue(result.anomalies.isEmpty())
        assertTrue(result.healthRisks.isEmpty())
    }

    @Test
    fun `analyze should return regularity score for 3 cycles`() {
        val cycles = listOf(
            createCycle(daysAgo = 28, cycleLength = 28),
            createCycle(daysAgo = 56, cycleLength = 29),
            createCycle(daysAgo = 84, cycleLength = 28)
        )

        val result = engine.analyze(cycles)

        assertNotNull(result.regularityScore)
        assertTrue(result.regularityScore!!.score > 70)
        assertEquals(RegularityLevel.VERY_REGULAR, result.regularityScore!!.level)
    }

    @Test
    fun `analyze should detect anomaly for irregular cycle`() {
        val cycles = listOf(
            createCycle(daysAgo = 28, cycleLength = 28),
            createCycle(daysAgo = 56, cycleLength = 29),
            createCycle(daysAgo = 84, cycleLength = 45),
            createCycle(daysAgo = 112, cycleLength = 28)
        )

        val result = engine.analyze(cycles)

        assertTrue(result.anomalies.isNotEmpty())
        assertEquals(AnomalyType.TOO_LONG, result.anomalies[0].anomalyType)
    }

    @Test
    fun `analyze should assess anemia risk for heavy periods`() {
        val cycles = listOf(
            createCycle(daysAgo = 28, cycleLength = 28, flowIntensity = "heavy"),
            createCycle(daysAgo = 56, cycleLength = 29, flowIntensity = "heavy"),
            createCycle(daysAgo = 84, cycleLength = 28, flowIntensity = "heavy"),
            createCycle(daysAgo = 112, cycleLength = 28, flowIntensity = "heavy")
        )

        val result = engine.analyze(cycles)

        assertTrue(result.healthRisks.any { it.riskType == HealthRiskType.ANEMIA })
    }

    private fun createCycle(daysAgo: Int, cycleLength: Int, flowIntensity: String = "medium"): MenstrualCycle {
        val calendar = Calendar.getInstance()
        calendar.add(Calendar.DAY_OF_MONTH, -daysAgo)
        
        return MenstrualCycle(
            id = 0,
            startDate = calendar.timeInMillis,
            endDate = calendar.timeInMillis + cycleLength * 24 * 60 * 60 * 1000L,
            cycleLength = cycleLength,
            flowIntensity = flowIntensity,
            symptoms = null,
            notes = null
        )
    }
}
```

## 九、常见问题

### Q1: 为什么需要至少3个周期数据？
A: 统计学上，至少需要3个数据点才能计算标准差和变异系数，从而评估周期的一致性。

### Q2: 趋势分析需要多少数据？
A: 趋势分析需要至少4个周期数据才能进行线性回归分析，建议提供6-12个周期数据以获得更准确的结果。

### Q3: 如何处理异常数据？
A: 分析引擎会自动检测异常周期（超出2个标准差），并根据偏差程度给出不同的建议。

### Q4: 健康风险评估的准确性如何？
A: 健康风险评估基于统计学方法和医学指南，仅供参考，不能替代专业医疗诊断。如需准确评估，请咨询专业医生。

### Q5: 分析结果多久更新一次？
A: 每次记录新周期后自动更新，用户也可以手动刷新分析结果。
