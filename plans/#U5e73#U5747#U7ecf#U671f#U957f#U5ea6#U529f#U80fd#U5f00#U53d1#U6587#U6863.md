## 一、功能概述

### 1.1 功能定义
平均经期长度功能是女性健康管理应用的基础统计模块，通过分析用户历史经期数据，计算平均经期长度，展示经期长度分布和趋势，帮助用户了解自身经期的正常状况。

### 1.2 核心特性
- **平均经期计算**：基于历史数据计算平均经期长度
- **经期分布统计**：展示不同经期长度的分布情况
- **正常范围判断**：基于医学标准判断经期是否正常
- **趋势追踪**：记录经期长度的变化趋势
- **异常预警**：识别过短或过长的经期

### 1.3 技术亮点
- 基于医学标准的正常范围（3-7天）
- 智能数据清洗，排除异常值
- 支持多种可视化方式
- 与其他周期分析功能无缝集成

## 二、经期长度标准

### 2.1 医学标准

| 经期长度 | 分类 | 颜色 | 说明 |
|---------|------|------|------|
| < 3天 | 过短 | #F44336 | 可能提示激素水平异常或子宫内膜问题 |
| 3-7天 | 正常 | #4CAF50 | 正常范围，健康指标良好 |
| 8-14天 | 偏长 | #FF9800 | 需要关注，可能与内分泌失调有关 |
| > 14天 | 过长 | #F44336 | 异常情况，强烈建议就医 |

### 2.2 统计数据

**全球平均经期长度**：4.6天  
**正常范围**：2-8天（95%的女性）  
**最常见**：3-5天  
**平均变化**：±1-2天属于正常波动

### 2.3 影响因素

- **年龄**：青春期和围绝经期经期可能不稳定
- **激素水平**：雌激素、孕激素水平影响
- **压力**：精神压力可能导致经期变化
- **营养状况**：营养不良或过度节食
- **药物**：避孕药、激素类药物
- **疾病**：多囊卵巢综合征、甲状腺功能异常等

### 步骤2: 创建经期长度分析引擎

**文件路径**: `domain/PeriodDurationAnalyzer.kt`
```kotlin
package com.example.womenhealthapp.domain

import com.example.womenhealthapp.data.entity.MenstrualCycle
import com.example.womenhealthapp.data.model.*
import kotlin.math.abs
import kotlin.math.sqrt

/**
 * 经期长度分析引擎
 */
class PeriodDurationAnalyzer(
    private val minCycles: Int = 3
) {

    /**
     * 分析经期长度统计
     * @param cycles 周期列表
     * @return 统计结果
     */
    fun analyze(cycles: List<MenstrualCycle>): PeriodDurationStatistics {
        // 1. 数据筛选
        val validCycles = cycles.filter { 
            it.periodDuration != null && 
            it.periodDuration!! >= 1 && 
            it.periodDuration!! <= 30
        }

        // 2. 数据验证
        if (validCycles.size < minCycles) {
            return createInsufficientDataResult(validCycles.size)
        }

        // 3. 计算统计数据
        val durations = validCycles.mapNotNull { it.periodDuration }.map { it.toFloat() }
        
        val averageDuration = durations.average().toFloat()
        val medianDuration = calculateMedian(durations)
        val minDuration = durations.minOrNull()?.toInt() ?: 0
        val maxDuration = durations.maxOrNull()?.toInt() ?: 0
        val stdDeviation = calculateStandardDeviation(durations, averageDuration)

        // 4. 分类分布统计
        val categoryDistribution = durations
            .map { PeriodDurationCategory.fromDuration(it.toInt()) }
            .groupBy { it }
            .mapValues { it.value.size }

        val mostCommonCategory = categoryDistribution.maxByOrNull { it.value }?.key 
            ?: PeriodDurationCategory.INSUFFICIENT_DATA

        // 5. 趋势分析
        val trend = analyzeTrend(validCycles)

        // 6. 生成洞察
        val insights = generateInsights(
            averageDuration,
            medianDuration,
            stdDeviation,
            mostCommonCategory,
            trend,
            validCycles.size
        )

        return PeriodDurationStatistics(
            averageDuration = averageDuration,
            medianDuration = medianDuration,
            minDuration = minDuration,
            maxDuration = maxDuration,
            stdDeviation = stdDeviation,
            totalCycles = validCycles.size,
            categoryDistribution = categoryDistribution,
            mostCommonCategory = mostCommonCategory,
            trend = trend,
            insights = insights
        )
    }

    /**
     * 计算中位数
     */
    private fun calculateMedian(values: List<Float>): Float {
        if (values.isEmpty()) return 0f

        val sorted = values.sorted()
        val size = sorted.size

        return if (size % 2 == 0) {
            (sorted[size / 2 - 1] + sorted[size / 2]) / 2f
        } else {
            sorted[size / 2]
        }
    }

    /**
     * 计算标准差
     */
    private fun calculateStandardDeviation(values: List<Float>, mean: Float): Float {
        if (values.isEmpty()) return 0f

        val variance = values.map { (it - mean).pow(2) }.average()
        return sqrt(variance).toFloat()
    }

    /**
     * 分析趋势
     */
    private fun analyzeTrend(cycles: List<MenstrualCycle>): DurationTrend {
        if (cycles.size < 4) return DurationTrend.INSUFFICIENT

        val durations = cycles.takeLast(4).mapNotNull { it.periodDuration }
        
        if (durations.size < 4) return DurationTrend.INSUFFICIENT

        // 计算变化
        val firstHalf = durations.take(2).average()
        val secondHalf = durations.drop(2).average()
        val change = secondHalf - firstHalf

        // 判断趋势
        return when {
            abs(change) < 0.5 -> DurationTrend.STABLE
            change > 1.0 -> DurationTrend.INCREASING
            change < -1.0 -> DurationTrend.DECREASING
            change > 0 -> DurationTrend.INCREASING
            else -> DurationTrend.DECREASING
        }
    }

    /**
     * 生成洞察
     */
    private fun generateInsights(
        average: Float,
        median: Float,
        stdDev: Float,
        mostCommon: PeriodDurationCategory,
        trend: DurationTrend,
        cycleCount: Int
    ): List<String> {
        val insights = mutableListOf<String>()

        // 基本统计
        insights.add("您的平均经期长度为${String.format("%.1f", average)}天")
        insights.add("中位数为${String.format("%.1f", median)}天")
        insights.add("基于${cycleCount}个周期的数据分析")

        // 正常性判断
        when {
            average in 3f..8f -> {
                insights.add("您的经期长度在正常范围内")
                insights.add("正常范围：3-7天")
            }
            average < 3f -> {
                insights.add("您的经期长度偏短，建议关注")
                insights.add("可能原因：激素水平异常、营养不良等")
            }
            average > 8f -> {
                insights.add("您的经期长度偏长，建议关注")
                insights.add("可能原因：内分泌失调、子宫肌瘤等")
            }
        }

        // 最常见分类
        insights.add("最常见的经期长度为：${mostCommon.displayName}")

        // 趋势信息
        if (trend != DurationTrend.INSUFFICIENT) {
            when (trend) {
                DurationTrend.STABLE -> {
                    insights.add("您的经期长度保持稳定")
                }
                DurationTrend.INCREASING -> {
                    insights.add("您的经期长度有增长趋势")
                    insights.add("可能与年龄、激素变化有关")
                }
                DurationTrend.DECREASING -> {
                    insights.add("您的经期长度有减少趋势")
                    insights.add("可能与压力、营养状况有关")
                }
                DurationTrend.FLUCTUATING -> {
                    insights.add("您的经期长度波动较大")
                    insights.add("建议记录更多细节，寻找波动原因")
                }
                else -> {}
            }
        }

        // 标准差信息
        if (stdDev > 1.5f) {
            insights.add("经期长度波动较大（标准差：${String.format("%.2f", stdDev)}天）")
            insights.add("正常波动应在±1-2天范围内")
        }

        // 医学建议
        when {
            average < 2f || average > 10f -> {
                insights.add("强烈建议咨询妇科医生")
            }
            average < 3f || average > 8f -> {
                insights.add("建议就医咨询，排除潜在问题")
            }
            else -> {
                insights.add("继续保持良好的生活习惯")
            }
        }

        return insights
    }

    /**
     * 创建数据不足结果
     */
    private fun createInsufficientDataResult(cycleCount: Int): PeriodDurationStatistics {
        return PeriodDurationStatistics(
            averageDuration = 0f,
            medianDuration = 0f,
            minDuration = 0,
            maxDuration = 0,
            stdDeviation = 0f,
            totalCycles = cycleCount,
            categoryDistribution = emptyMap(),
            mostCommonCategory = PeriodDurationCategory.INSUFFICIENT_DATA,
            trend = DurationTrend.INSUFFICIENT,
            insights = listOf(
                "需要至少${minCycles}个周期数据才能分析",
                "当前已有${cycleCount}个周期数据，请继续记录",
                "记录经期开始和结束日期以获得经期长度"
            )
        )
    }
}
```

```kotlin
                WHEN periodDuration BETWEEN 7 AND 8 THEN 'NORMAL_LONG'
                WHEN periodDuration BETWEEN 9 AND 14 THEN 'LONG'
                ELSE 'TOO_LONG'
            END as category,
            COUNT(*) as count
        FROM menstrual_cycles 
        WHERE periodDuration IS NOT NULL
        GROUP BY category
    """)
```

```kotlin
        menstrualCycleDao.getAverageCycleLength()?.toInt()
    
    fun getCyclesByDateRange(startDate: Long, endDate: Long): Flow<List<MenstrualCycle>> = 
        menstrualCycleDao.getCyclesByDateRange(startDate, endDate)

    // ========== 新增方法：经期长度分析 ==========

     * 分析经期长度统计
     */

        val cycles = menstrualCycleDao.getCyclesWithPeriodDuration()
        return periodDurationAnalyzer.analyze(cycles)
    }
    
    /**
     * 获取平均经期天数
     */
    suspend fun getAveragePeriodDuration(): Float? {
        return menstrualCycleDao.getAveragePeriodDuration()
    }
    
    /**
     * 计算周期长度（用于自动计算periodDuration）
     */
    fun calculatePeriodDuration(startDate: Long, endDate: Long): Int {
        val diffMs = endDate - startDate
        return (diffMs / (24 * 60 * 60 * 1000)).toInt() + 1 // 包含首尾两天
    }
}
```

```kotlin
package com.example.womenhealthapp.viewmodel

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope

import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class PeriodDurationViewModel(
    private val repository: WomenHealthRepository
) : ViewModel() {

    private val _statistics = MutableStateFlow<PeriodDurationStatistics?>(null)
    val statistics: StateFlow<PeriodDurationStatistics?> = _statistics.asStateFlow()

    private val _isLoading = MutableStateFlow(false)
    val isLoading: StateFlow<Boolean> = _isLoading.asStateFlow()

    private val _error = MutableStateFlow<String?>(null)
    val error: StateFlow<String?> = _error.asStateFlow()

    /**
     * 加载经期长度统计
     */
    fun loadStatistics() {
        viewModelScope.launch {
            try {
                _isLoading.value = true
                _error.value = null

                val stats = repository.analyzePeriodDuration()
                _statistics.value = stats
            } catch (e: Exception) {
                _error.value = "加载统计失败: ${e.message}"
            } finally {
                _isLoading.value = false
            }
        }
    }

    /**
     * 刷新统计
     */
    fun refresh() {
        loadStatistics()
    }

    /**
     * 清除错误信息
     */
    fun clearError() {
        _error.value = null
    }
}
```

```kotlin
import kotlinx.coroutines.launch

class PeriodDurationFragment : Fragment() {

    private val viewModel: PeriodDurationViewModel by viewModels()

    private lateinit var averageDurationTextView: TextView
    private lateinit var categoryBadgeView: View
    private lateinit var categoryIconView: TextView
    private lateinit var categoryTextView: TextView
    private lateinit var trendTextView: TextView
    private lateinit var statsLayout: LinearLayout
    private lateinit var distributionLayout: LinearLayout
    private lateinit var insightsLayout: LinearLayout

    override fun onCreateView(
        inflater: LayoutInflater,
        container: ViewGroup?,
        savedInstanceState: Bundle?
    ): View? {
        return inflater.inflate(R.layout.fragment_period_duration, container, false)
    }

    override fun onViewCreated(view: View, savedInstanceState: Bundle?) {
        super.onViewCreated(view, savedInstanceState)
        
        initViews(view)
        observeViewModel()
        viewModel.loadStatistics()
    }

    private fun initViews(view: View) {
        averageDurationTextView = view.findViewById(R.id.averageDurationTextView)
        categoryBadgeView = view.findViewById(R.id.categoryBadgeView)
        categoryIconView = view.findViewById(R.id.categoryIconView)
        categoryTextView = view.findViewById(R.id.categoryTextView)
        trendTextView = view.findViewById(R.id.trendTextView)
        statsLayout = view.findViewById(R.id.statsLayout)
        distributionLayout = view.findViewById(R.id.distributionLayout)
        insightsLayout = view.findViewById(R.id.insightsLayout)
    }

    private fun observeViewModel() {
        viewLifecycleOwner.lifecycleScope.launch {
            viewModel.statistics.collect { stats ->
                stats?.let { updateUI(it) }
            }
        }

        viewLifecycleOwner.lifecycleScope.launch {
            viewModel.isLoading.collect { isLoading ->
                // 显示/隐藏加载状态
            }
        }

        viewLifecycleOwner.lifecycleScope.launch {
            viewModel.error.collect { error ->
                error?.let { 
                    // 显示错误信息
                    viewModel.clearError()
                }
            }
        }
    }

    private fun updateUI(stats: com.example.womenhealthapp.data.model.PeriodDurationStatistics) {
        // 更新平均经期长度
        averageDurationTextView.text = String.format("%.1f", stats.averageDuration)

        // 更新分类徽章
        val category = PeriodDurationCategory.fromDuration(stats.averageDuration.toInt())
        categoryIconView.text = category.icon
        categoryTextView.text = category.displayName
        categoryTextView.setTextColor(Color.parseColor(category.color))
        categoryBadgeView.setBackgroundColor(Color.parseColor(category.backgroundColor))

        // 更新趋势
        trendTextView.text = "${stats.trend.icon} ${stats.trend.displayName}"

        // 更新统计数据
        updateStats(stats)

        // 更新分布
        updateDistribution(stats.categoryDistribution)

        // 更新洞察
        updateInsights(stats.insights)
    }

    private fun updateStats(stats: com.example.womenhealthapp.data.model.PeriodDurationStatistics) {
        statsLayout.removeAllViews()

        val statsList = listOf(
            "中位数" to "${String.format("%.1f", stats.medianDuration)}天",
            "最短经期" to "${stats.minDuration}天",
            "最长经期" to "${stats.maxDuration}天",
            "标准差" to String.format("%.2f天", stats.stdDeviation),
            "数据周期" to "${stats.totalCycles}个"
        )

        for ((label, value) in statsList) {
            val view = createStatView(label, value)
            statsLayout.addView(view)
        }
    }

    private fun createStatView(label: String, value: String): View {
        val view = layoutInflater.inflate(R.layout.item_stat_small, null)
        
        val labelTextView = view.findViewById<TextView>(R.id.labelTextView)
        val valueTextView = view.findViewById<TextView>(R.id.valueTextView)

        labelTextView.text = label
        valueTextView.text = value

        return view
    }

    private fun updateDistribution(distribution: Map<PeriodDurationCategory, Int>) {
        distributionLayout.removeAllViews()

        if (distribution.isEmpty()) {
            val noDataView = createNoDataView()
            distributionLayout.addView(noDataView)
        } else {
            // 按优先级排序
            distribution.entries
                .sortedByDescending { it.key.priority }
                .forEach { (category, count) ->
                    val view = createDistributionView(category, count, distribution.values.sum())
                    distributionLayout.addView(view)
                }
        }
    }

    private fun createNoDataView(): View {
        val view = layoutInflater.inflate(R.layout.item_no_data, null)
        val textView = view.findViewById<TextView>(R.id.messageTextView)
        textView.text = "暂无数据"
        return view
    }

    private fun createDistributionView(
        category: PeriodDurationCategory,
        count: Int,
        total: Int
    ): View {
        val view = layoutInflater.inflate(R.layout.item_duration_distribution, null)
        
        val iconTextView = view.findViewById<TextView>(R.id.iconTextView)
        val nameTextView = view.findViewById<TextView>(R.id.nameTextView)
        val countTextView = view.findViewById<TextView>(R.id.countTextView)
        val percentageTextView = view.findViewById<TextView>(R.id.percentageTextView)
        val progressBar = view.findViewById<ProgressBar>(R.id.progressBar)

        iconTextView.text = category.icon
        nameTextView.text = category.displayName
        nameTextView.setTextColor(Color.parseColor(category.color))
        countTextView.text = "$count次"
        val percentage = if (total > 0) (count * 100 / total) else 0
        percentageTextView.text = "$percentage%"
        progressBar.max = total
        progressBar.progress = count
        progressBar.progressTintList = android.content.res.ColorStateList.valueOf(
            Color.parseColor(category.color)
        )

        return view
    }

    private fun updateInsights(insights: List<String>) {
        insightsLayout.removeAllViews()

        insights.forEach { insight ->
            val view = createInsightView(insight)
            insightsLayout.addView(view)
        }
    }

    private fun createInsightView(text: String): View {
        val view = layoutInflater.inflate(R.layout.item_insight, null)
        
        val textView = view.findViewById<TextView>(R.id.insightTextView)
        textView.text = "• $text"

        return view
    }
}
```

### 6.1 单元测试

**文件路径**: `domain/PeriodDurationAnalyzerTest.kt`
```kotlin
package com.example.womenhealthapp.domain

import com.example.womenhealthapp.data.entity.MenstrualCycle
import com.example.womenhealthapp.data.model.PeriodDurationCategory
import org.junit.Assert.*
import org.junit.Before
import org.junit.Test
import java.util.Calendar

class PeriodDurationAnalyzerTest {

    private lateinit var analyzer: PeriodDurationAnalyzer

    @Before
    fun setup() {
        analyzer = PeriodDurationAnalyzer()
    }

    @Test
    fun `analyze should return insufficient data when cycles less than 3`() {
        val cycles = listOf(
            createCycle(daysAgo = 28, periodDuration = 5),
            createCycle(daysAgo = 56, periodDuration = 6)
        )

        val result = analyzer.analyze(cycles)

        assertEquals(0f, result.averageDuration)
        assertEquals(0, result.totalCycles)
    }

    @Test
    fun `analyze should calculate correct average for normal periods`() {
        val cycles = listOf(
            createCycle(daysAgo = 28, periodDuration = 5),
            createCycle(daysAgo = 56, periodDuration = 6),
            createCycle(daysAgo = 84, periodDuration = 5),
            createCycle(daysAgo = 112, periodDuration = 6),
            createCycle(daysAgo = 140, periodDuration = 5)
        )

        val result = analyzer.analyze(cycles)

        assertEquals(5.4f, result.averageDuration, 0.1f)
        assertEquals(PeriodDurationCategory.NORMAL_MEDIUM, result.mostCommonCategory)
        assertTrue(result.averageDuration in 3f..8f)
    }

    @Test
    fun `analyze should detect too short periods`() {
        val cycles = listOf(
            createCycle(daysAgo = 28, periodDuration = 2),
            createCycle(daysAgo = 56, periodDuration = 3),
            createCycle(daysAgo = 84, periodDuration = 2)
        )

        val result = analyzer.analyze(cycles)

        assertTrue(result.averageDuration < 3f)
        assertEquals(PeriodDurationCategory.TOO_SHORT, result.mostCommonCategory)
    }

    @Test
    fun `analyze should detect too long periods`() {
        val cycles = listOf(
            createCycle(daysAgo = 28, periodDuration = 10),
            createCycle(daysAgo = 56, periodDuration = 12),
            createCycle(daysAgo = 84, periodDuration = 11)
        )

        val result = analyzer.analyze(cycles)

        assertTrue(result.averageDuration > 8f)
        assertEquals(PeriodDurationCategory.LONG, result.mostCommonCategory)
    }

    private fun createCycle(daysAgo: Int, periodDuration: Int): MenstrualCycle {
        val calendar = Calendar.getInstance()
        calendar.add(Calendar.DAY_OF_MONTH, -daysAgo)
        
        return MenstrualCycle(
            id = 0,
            startDate = calendar.timeInMillis,
            endDate = calendar.timeInMillis + 28 * 24 * 60 * 60 * 1000L,
            cycleLength = 28,
            periodDuration = periodDuration,
            flowIntensity = "medium",
            symptoms = null,
            notes = null
        )
    }
}
```

## 十一、常见问题

### Q1: 正常的经期长度是多少？
A: 正常的经期长度是3-7天，平均约4.6天。2-8天也属于正常范围。

### Q2: 经期过短或过长怎么办？
A: 经期过短（<3天）或过长（>7天）都可能提示健康问题，建议咨询妇科医生。

### Q3: 影响经期长度的因素有哪些？
A: 年龄、激素水平、压力、营养状况、药物、疾病等都可能影响经期长度。

### Q4: 经期长度会变化吗？
A: 会。青春期和围绝经期经期可能不稳定，正常情况下也会有±1-2天的波动。

### Q5: 如何记录经期长度？
A: 记录经期的开始和结束日期，系统会自动计算经期长度。
