## 一、功能概述

### 1.1 功能定义
周期规律性评分功能是女性健康管理应用的核心功能之一，通过分析用户的历史经期数据，计算出一个综合评分（0-100分），帮助用户了解自身月经周期的规律性和健康状况。

### 1.2 核心特性
- **多维度评估**: 从周期一致性、正常性、经期稳定性、数据充足性四个维度综合评分
- **健康预警**: 自动识别周期异常，提供健康建议
- **趋势分析**: 展示评分的历史变化趋势
- **个性化建议**: 根据评分等级提供针对性的健康建议

### 1.3 技术亮点
- 本地计算，保护用户隐私
- 智能数据清洗，剔除异常值
- 支持实时更新和历史追溯
- 算法基于医学统计学标准

```kotlin
data class RegularityScore(
    val score: Int,                           // 综合评分（0-100）
    val level: ScoreLevel,                     // 评分等级
    val dimensions: DimensionScores,           // 各维度评分

data class DimensionScores(

    val stability: Int,        // 经期稳定性（权重20%）
    val dataSufficiency: Int   // 数据充足性（权重10%）
) {
    val totalScore: Int
        get() = consistency + normality + stability + dataSufficiency
}

data class CycleStatistics(
    val averageCycleLength: Float,             // 平均周期长度
    val cycleStdDev: Float,                    // 周期标准差

    val totalCycleCount: Int,                  // 总周期数量
    val trend: CycleTrend                      // 周期趋势
)

enum class CycleTrend {
    STABLE,        // 稳定
    IMPROVING,     // 改善
    WORSENING,     // 恶化
    INSUFFICIENT   // 数据不足
}
```

### 步骤2: 创建评分计算器

**文件路径**: `domain/RegularityScoreCalculator.kt`
```kotlin
package com.example.womenhealthapp.domain

import com.example.womenhealthapp.data.entity.MenstrualCycle
import com.example.womenhealthapp.data.model.*
import kotlin.math.abs
import kotlin.math.pow
import kotlin.math.sqrt

class RegularityScoreCalculator(
    private val minCycles: Int = 3,
    private val maxCycles: Int = 12
) {

    /**
     * 计算规律性评分
     * @param cycles 周期记录列表
     * @return 评分结果
     */
    fun calculate(cycles: List<MenstrualCycle>): RegularityScore {
        // 1. 数据筛选
        val filteredCycles = filterValidCycles(cycles)
        val recentCycles = selectRecentCycles(filteredCycles)

        // 2. 数据验证
        if (recentCycles.size < minCycles) {
            return createInsufficientDataResult()
        }

        // 3. 计算统计数据
        val statistics = calculateStatistics(recentCycles)

        // 4. 计算各维度评分
        val dimensions = calculateDimensionScores(statistics, recentCycles.size)

        // 5. 确定评分等级
        val level = ScoreLevel.fromScore(dimensions.totalScore)

        // 6. 生成建议
        val recommendations = generateRecommendations(level, statistics)

        return RegularityScore(
            score = dimensions.totalScore,
            level = level,
            dimensions = dimensions,
            statistics = statistics,
            recommendations = recommendations
        )
    }

    /**
     * 筛选有效的周期数据
     */
    private fun filterValidCycles(cycles: List<MenstrualCycle>): List<MenstrualCycle> {
        return cycles.filter { cycle ->
            cycle.cycleLength in 18..45 // 正常周期范围
        }
    }

    /**
     * 选择最近的N个周期
     */
    private fun selectRecentCycles(cycles: List<MenstrualCycle>): List<MenstrualCycle> {
        return cycles
            .sortedByDescending { it.startDate }
            .take(maxCycles)
            .sortedBy { it.startDate }
    }

    /**
     * 计算统计数据
     */
    private fun calculateStatistics(cycles: List<MenstrualCycle>): CycleStatistics {
        val cycleLengths = cycles.map { it.cycleLength.toFloat() }
        
        val avgCycleLength = cycleLengths.average().toFloat()
        val cycleStdDev = calculateStandardDeviation(cycleLengths, avgCycleLength)
        val cv = if (avgCycleLength > 0) cycleStdDev / avgCycleLength else 0f
        
        val abnormalCount = cycles.count { it.cycleLength !in 21..35 }
        val trend = analyzeTrend(cycles)

        return CycleStatistics(
            averageCycleLength = avgCycleLength,
            cycleStdDev = cycleStdDev,
            cycleCoefficientOfVariation = cv,
            abnormalCycleCount = abnormalCount,
            totalCycleCount = cycles.size,
            trend = trend
        )
    }

    /**
     * 计算标准差
     */
    private fun calculateStandardDeviation(values: List<Float>, mean: Float): Float {
        if (values.isEmpty()) return 0f
        
        val variance = values.map { (it - mean).pow(2) }.average()
        return sqrt(variance).toFloat()
    }

    /**
     * 分析周期趋势
     */
    private fun analyzeTrend(cycles: List<MenstrualCycle>): CycleTrend {
        if (cycles.size < 3) return CycleTrend.INSUFFICIENT

        val recentLengths = cycles.takeLast(3).map { it.cycleLength }
        val avgRecent = recentLengths.average()
        val avgPrevious = cycles.dropLast(3).map { it.cycleLength }.average()

        return when {
            recentLengths[2] > recentLengths[1] && recentLengths[1] > recentLengths[0] -> 
                CycleTrend.IMPROVING
            recentLengths[2] < recentLengths[1] && recentLengths[1] < recentLengths[0] -> 
                CycleTrend.WORSENING
            abs(avgRecent - avgPrevious) < 2 -> CycleTrend.STABLE
            avgRecent > avgPrevious -> CycleTrend.WORSENING
            else -> CycleTrend.IMPROVING
        }
    }

    /**
     * 计算各维度评分
     */
    private fun calculateDimensionScores(
        statistics: CycleStatistics,
        cycleCount: Int
    ): DimensionScores {
        return DimensionScores(
            consistency = calculateConsistencyScore(statistics),
            normality = calculateNormalityScore(statistics),
            stability = calculateStabilityScore(statistics),
            dataSufficiency = calculateDataScore(cycleCount)
        )
    }

    /**
     * 计算周期一致性评分（权重40%）
     */
    private fun calculateConsistencyScore(statistics: CycleStatistics): Int {
        return when (statistics.cycleCoefficientOfVariation) {
            in 0f..0.07f -> 40      // 非常规律
            in 0.07f..0.15f -> 30   // 规律
            in 0.15f..0.25f -> 20   // 轻微不规律
            in 0.25f..0.35f -> 10   // 不规律
            else -> 5               // 非常不规律
        }
    }

    /**
     * 计算周期正常性评分（权重30%）
     */
    private fun calculateNormalityScore(statistics: CycleStatistics): Int {
        return when {
            statistics.averageCycleLength in 25f..30f -> 30   // 非常正常
            statistics.averageCycleLength in 21f..25f || 
            statistics.averageCycleLength in 30f..35f -> 25   // 正常
            statistics.averageCycleLength in 18f..21f || 
            statistics.averageCycleLength in 35f..45f -> 15   // 异常
            else -> 5                                           // 严重异常
        }
    }

    /**
     * 计算经期稳定性评分（权重20%）
     * 注意：当前MenstrualCycle实体缺少periodDuration字段
     * 如需实现，需要先扩展数据模型
     */
    private fun calculateStabilityScore(statistics: CycleStatistics): Int {
        // 暂时使用周期标准差作为替代指标
        return when (statistics.cycleStdDev) {
            in 0f..2f -> 20
            in 2f..4f -> 15
            in 4f..6f -> 10
            else -> 5
        }
    }

    /**
     * 计算数据充足性评分（权重10%）
     */
    private fun calculateDataScore(cycleCount: Int): Int {
        return when (cycleCount) {
            in 12..Int.MAX_VALUE -> 10
            in 6..11 -> 8
            in 3..5 -> 5
            else -> 0
        }
    }

    /**
     * 生成健康建议
     */
    private fun generateRecommendations(
        level: ScoreLevel,
        statistics: CycleStatistics
    ): List<String> {
        val recommendations = mutableListOf<String>()

        when (level) {
            ScoreLevel.EXCELLENT -> {
                recommendations.add("您的周期非常规律，继续保持健康的生活方式")
                recommendations.add("建议定期记录周期数据，及时发现细微变化")
            }
            ScoreLevel.GOOD -> {
                recommendations.add("您的周期规律，继续保持良好的生活习惯")
                recommendations.add("保持规律作息，均衡饮食，适度运动")
            }
            ScoreLevel.FAIR -> {
                recommendations.add("您的周期轻微不规律，建议关注以下几点：")
                if (statistics.cycleCoefficientOfVariation > 0.15) {
                    recommendations.add("- 保持规律的作息时间")
                    recommendations.add("- 减少压力，适当运动")
                    recommendations.add("- 避免过度节食或暴饮暴食")
                }
                if (statistics.abnormalCycleCount > 0) {
                    recommendations.add("- 注意周期长度的变化")
                }
            }
            ScoreLevel.POOR -> {
                recommendations.add("您的周期不规律，建议：")
                recommendations.add("- 记录详细的生活习惯（饮食、运动、睡眠）")
                recommendations.add("- 避免过度节食或暴饮暴食")
                recommendations.add("- 如持续不规律，建议就医检查")
                if (statistics.abnormalCycleCount > 0) {
                    recommendations.add("- 您的周期长度超出正常范围，请重点关注")
                }
            }
            ScoreLevel.VERY_POOR -> {
                recommendations.add("您的周期非常不规律，强烈建议：")
                recommendations.add("- 尽快就医检查内分泌功能")
                recommendations.add("- 可能的疾病：多囊卵巢综合征、甲状腺功能异常等")
                recommendations.add("- 记录详细的症状和生活日志，帮助医生诊断")
            }
            ScoreLevel.INSUFFICIENT_DATA -> {
                recommendations.add("请继续记录至少3个周期数据以获得准确评分")
                recommendations.add("建议记录周期开始和结束日期")
            }
        }

        return recommendations
    }

    /**
     * 创建数据不足的结果
     */
    private fun createInsufficientDataResult(): RegularityScore {
        return RegularityScore(
            score = 0,
            level = ScoreLevel.INSUFFICIENT_DATA,
            dimensions = DimensionScores(0, 0, 0, 0),
            statistics = CycleStatistics(
                averageCycleLength = 0f,
                cycleStdDev = 0f,
                cycleCoefficientOfVariation = 0f,
                abnormalCycleCount = 0,
                totalCycleCount = 0,
                trend = CycleTrend.INSUFFICIENT
            ),
            recommendations = listOf(
                "请继续记录至少3个周期数据以获得准确评分",
                "建议记录周期开始和结束日期"
            )
        )
    }
}
```

```kotlin
package com.example.womenhealthapp.viewmodel

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.example.womenhealthapp.data.model.RegularityScore
import com.example.womenhealthapp.repository.WomenHealthRepository
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.launch

class RegularityScoreViewModel(
    private val repository: WomenHealthRepository
) : ViewModel() {

    private val _scoreState = MutableStateFlow<RegularityScore?>(null)
    val scoreState: StateFlow<RegularityScore?> = _scoreState.asStateFlow()

    private val _isLoading = MutableStateFlow(false)
    val isLoading: StateFlow<Boolean> = _isLoading.asStateFlow()

    private val _error = MutableStateFlow<String?>(null)
    val error: StateFlow<String?> = _error.asStateFlow()

    /**
     * 加载当前评分
     */
    fun loadCurrentScore() {
        viewModelScope.launch {
            try {
                _isLoading.value = true
                _error.value = null

                val score = repository.calculateRegularityScore()
                _scoreState.value = score
            } catch (e: Exception) {
                _error.value = "加载评分失败: ${e.message}"
            } finally {
                _isLoading.value = false
            }
        }
    }

    /**
     * 刷新评分
     */
    fun refreshScore() {
        loadCurrentScore()
    }

    /**
     * 清除错误信息
     */
    fun clearError() {
        _error.value = null
    }
}
```

```kotlin
class RegularityScoreFragment : Fragment() {

    private val viewModel: RegularityScoreViewModel by viewModels()

    private lateinit var scoreCircleView: View

    private lateinit var levelTextView: TextView
    private lateinit var descriptionTextView: TextView

    private fun initViews(view: View) {

            }
        }

            viewModel.isLoading.collect { isLoading ->

            viewModel.error.collect { error ->

    private fun updateUI(score: com.example.womenhealthapp.data.model.RegularityScore) {

        // 更新统计数据
        updateStatistics(score.statistics)

        // 更新健康建议
        updateRecommendations(score.recommendations)
    }

    private fun updateDimensionScores(dimensions: com.example.womenhealthapp.data.model.DimensionScores) {
        dimensionsLayout.removeAllViews()

        val dimensionsList = listOf(
            Pair("周期一致性", dimensions.consistency, 40),
            Pair("周期正常性", dimensions.normality, 30),
            Pair("经期稳定性", dimensions.stability, 20),
            Pair("数据充足性", dimensions.dataSufficiency, 10)
        )

        for ((name, score, maxScore) in dimensionsList) {
            val view = createDimensionView(name, score, maxScore)
            dimensionsLayout.addView(view)
        }
```

```kotlin
package com.example.womenhealthapp.domain

import com.example.womenhealthapp.data.entity.MenstrualCycle
import com.example.womenhealthapp.data.model.ScoreLevel
import org.junit.Assert.*
import org.junit.Before
import org.junit.Test
import java.util.Calendar

class RegularityScoreCalculatorTest {

    private lateinit var calculator: RegularityScoreCalculator

    @Before
    fun setup() {
        calculator = RegularityScoreCalculator()
    }

    @Test
    fun `calculate should return insufficient data when cycles less than 3`() {
        val cycles = listOf(
            createCycle(daysAgo = 28, cycleLength = 28),
            createCycle(daysAgo = 56, cycleLength = 29)
        )

        val result = calculator.calculate(cycles)

        assertEquals(0, result.score)
        assertEquals(ScoreLevel.INSUFFICIENT_DATA, result.level)
    }

    @Test
    fun `calculate should return excellent score for regular cycles`() {
        val cycles = listOf(
            createCycle(daysAgo = 28, cycleLength = 28),
            createCycle(daysAgo = 56, cycleLength = 29),
            createCycle(daysAgo = 84, cycleLength = 28),
            createCycle(daysAgo = 112, cycleLength = 27),
            createCycle(daysAgo = 140, cycleLength = 28),
            createCycle(daysAgo = 168, cycleLength = 29)
        )

        val result = calculator.calculate(cycles)

        assertTrue(result.score >= 80)
        assertEquals(ScoreLevel.EXCELLENT, result.level)
    }

    @Test
    fun `calculate should return poor score for irregular cycles`() {
        val cycles = listOf(
            createCycle(daysAgo = 28, cycleLength = 21),
            createCycle(daysAgo = 56, cycleLength = 35),
            createCycle(daysAgo = 84, cycleLength = 40),
            createCycle(daysAgo = 112, cycleLength = 22)
        )

        val result = calculator.calculate(cycles)

        assertTrue(result.score < 70)
    }

    private fun createCycle(daysAgo: Int, cycleLength: Int): MenstrualCycle {
        val calendar = Calendar.getInstance()
        calendar.add(Calendar.DAY_OF_MONTH, -daysAgo)
        
        return MenstrualCycle(
            id = 0,
            startDate = calendar.timeInMillis,
            endDate = calendar.timeInMillis + cycleLength * 24 * 60 * 60 * 1000L,
            cycleLength = cycleLength,
            flowIntensity = "medium",
            symptoms = null,
            notes = null
        )
    }
}
```

## 九、常见问题

### Q1: 为什么需要至少3个周期数据？
A: 统计学上，至少需要3个数据点才能计算标准差和变异系数，从而评估周期的一致性。

### Q2: 评分标准的医学依据是什么？
A: 评分标准基于世界卫生组织（WHO）和妇科医学的周期正常性标准，周期长度正常范围为21-35天。

### Q3: 如何处理异常数据？
A: 计算器会自动过滤掉周期长度在18-45天范围外的数据，确保评分的准确性。

### Q4: 评分多久更新一次？
A: 每次记录新周期后自动更新，用户也可以手动刷新评分。
